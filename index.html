function download(){ const element = document.getElementById("pdf"); const opt = { margin: 0.5, filename: 'MS-Developer-House-Invoice.pdf', image: { type: 'jpeg', quality: 1 }, html2canvas: { scale: 3, useCORS: true, scrollY: 0 }, jsPDF: { unit: 'in', format: 'a4', orientation: 'portrait' } }; // small delay ensures full render setTimeout(() => { html2pdf().set(opt).from(element).save(); }, 300); }
